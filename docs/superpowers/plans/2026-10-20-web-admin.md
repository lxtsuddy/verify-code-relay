# Web 管理介面 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 verify-code-relay 的設定從 YAML 移到 SQLite，並提供一個對公網安全的 Web 管理介面（單一管理員、密碼＋TOTP），所有設定都能在網頁上完成、存檔即在程式內重載。

**Architecture:** 設定以「整份原始 YAML 文件＋版本」存在 `config_versions`；`internal/runtime` 持有目前運作中的服務並負責 `Apply` 重載；`internal/web` 以 Go `html/template`＋`embed` 提供伺服器渲染頁面與少量 JSON 端點；`internal/websec` 提供密碼雜湊、TOTP、QR、限速、用戶端 IP 等安全元件。Web 在 `:8080`，健康檢查移到 `:8081`。

**Tech Stack:** Go 1.26、`modernc.org/sqlite`、`gopkg.in/yaml.v3`、`golang.org/x/crypto/argon2`、`golang.org/x/term`、`rsc.io/qr`（新增）、`golang.org/x/net/html`（既有）、`go-imap/v2`。

**Spec:** `docs/superpowers/specs/2026-10-20-web-admin-design.md`

## Global Constraints

- 所有使用者可見文字、錯誤訊息、日誌訊息用繁體中文，跟隨既有風格。
- 新依賴只允許：`rsc.io/qr`。`golang.org/x/crypto`、`golang.org/x/term` 已在模組圖中，可直接改為直接依賴。不得引入 Node、前端框架、CDN。
- CSP：`default-src 'self'; img-src 'self' data:; style-src 'self'; script-src 'self'; frame-ancestors 'none'; form-action 'self'; base-uri 'none'`。所有 CSS／JS 都是 `embed` 的獨立檔案，**不得**使用 inline `<style>`、`style=` 屬性、inline `<script>`、`onclick` 等事件屬性；字型用系統字型。
- 機密（Bot token、應用程式密碼、TOTP 金鑰、session token）不得出現在任何頁面、JSON 回應、錯誤訊息或日誌中。
- 所有前端頁面（範本／CSS／JS）的任務，實作者**必須先叫用 frontend-design skill**，做出好看、互動完善、體驗好的介面；行動裝置寬度也要可用。
- 每個任務完成前必須通過：`gofmt -l .` 無輸出、`go vet ./...`、`go test -race -count=1 ./...`。
- Commit 訊息風格同 repo（`feat(web): ...`、`fix(store): ...`），結尾空一行加 `Co-Authored-By: Claude Code <noreply@anthropic.com>`。
- 工作分支：`web-admin`（從 master 開）。不要 push。
- 環境變數：`RELAY_DATA_DIR`（預設 `/data`）、`RELAY_LISTEN`（預設 `:8080`）、`RELAY_HEALTH_LISTEN`（預設 `:8081`，空字串不啟動）、`RELAY_ADMIN_PASSWORD`、`RELAY_TRUSTED_PROXIES`。
- 密碼最短 12 個字元（環境變數建立、Web 修改、`admin reset` 皆同）。
- Session：cookie `__Host-relay_session`（`HttpOnly`、`Secure`、`SameSite=Strict`、`Path=/`），閒置 30 分鐘、絕對 12 小時。
- 登入限速：全域 token bucket，每秒 1 次、容量 5。設定版本保留最近 50 版。

`encode.go`：

```go
package config

import (
	"bytes"
	"fmt"
	"os"

	"gopkg.in/yaml.v3"
)

// Empty returns a config with no bots, targets or mailboxes; it is valid once defaulted.
func Empty() *Config {
	return &Config{Bots: map[string]Bot{}, Targets: map[string]Target{}, Mailboxes: map[string]*Mailbox{}}
}

// Decode parses YAML exactly as stored: ${ENV} is not expanded and no defaults are applied.
func Decode(data []byte) (*Config, error) {
	var root yaml.Node
	if err := yaml.Unmarshal(data, &root); err != nil {
		return nil, fmt.Errorf("解析 YAML：%w", err)
	}
	c, err := decodeNode(&root)
	if err != nil {
		return nil, err
	}
	c.fillMaps()
	return c, nil
}

// Encode writes the config as YAML, leaving out every field that was not set.
func Encode(c *Config) ([]byte, error) {
	var buf bytes.Buffer
	enc := yaml.NewEncoder(&buf)
	enc.SetIndent(2)
	if err := enc.Encode(c); err != nil {
		return nil, err
	}
	if err := enc.Close(); err != nil {
		return nil, err
	}
	return buf.Bytes(), nil
}

// Clone returns a deep copy.
func (c *Config) Clone() *Config {
	data, err := Encode(c)
	if err != nil {
		panic(fmt.Sprintf("config: encode for clone: %v", err)) // every Config value is encodable
	}
	out, err := Decode(data)
	if err != nil {
		panic(fmt.Sprintf("config: decode for clone: %v", err))
	}
	return out
}

// Defaulted returns a copy with defaults applied; c is not modified.
func (c *Config) Defaulted() *Config {
	d := c.Clone()
	d.applyDefaults()
	return d
}

// LoadRaw reads a YAML file, expands ${ENV} and decodes it without applying defaults.
func LoadRaw(path string) (*Config, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, fmt.Errorf("讀取設定檔：%w", err)
	}
	return LoadRawBytes(data, os.LookupEnv)
}

// LoadRawBytes is LoadRaw for data already in memory.
func LoadRawBytes(data []byte, lookup func(string) (string, bool)) (*Config, error) {
	var root yaml.Node
	if err := yaml.Unmarshal(data, &root); err != nil {
		return nil, fmt.Errorf("解析 YAML：%w", err)
	}
	if err := expandNode(&root, "", lookup); err != nil {
		return nil, err
	}
	c, err := decodeNode(&root)
	if err != nil {
		return nil, err
	}
	c.fillMaps()
	return c, nil
}

// fillMaps initialises nil maps and sets Mailbox.Name, so callers can add entries directly.
func (c *Config) fillMaps() {
	if c.Bots == nil {
		c.Bots = map[string]Bot{}
	}
	if c.Targets == nil {
		c.Targets = map[string]Target{}
	}
	if c.Mailboxes == nil {
		c.Mailboxes = map[string]*Mailbox{}
	}
	for name, m := range c.Mailboxes {
		if m == nil {
			m = &Mailbox{}
			c.Mailboxes[name] = m
		}
		m.Name = name
	}
}
```

const (
	argonTime    = 2
	argonMemory  = 19 * 1024 // KiB (OWASP minimum for argon2id)
	argonThreads = 1
	argonKeyLen  = 32
)

func HashPassword(pw string) (string, error) {
	salt := make([]byte, 16)
	if _, err := rand.Read(salt); err != nil {
		return "", err
	}
	hashSem <- struct{}{}
	key := argon2.IDKey([]byte(pw), salt, argonTime, argonMemory, argonThreads, argonKeyLen)

}

	ip := websec.ClientIP(r, s.o.TrustedProxies)
	if !s.limiter.Allow() {
		s.renderLogin(w, r, http.StatusTooManyRequests, "嘗試太頻繁，請稍後再試")
		return
	}
	admin, ok, err := s.o.Store.GetAdmin(r.Context())
	if err != nil {
		s.serverError(w, r, err)
		return
	}
	if !ok {
		s.renderLogin(w, r, http.StatusServiceUnavailable, "尚未設定管理員：請設定 RELAY_ADMIN_PASSWORD 後重新啟動服務")
		return
	}
	pwOK, err := websec.VerifyPassword(admin.PasswordHash, r.PostFormValue("password"))
	if err != nil {
		s.serverError(w, r, err)
		return
	}
	if admin.TOTPSecret == "" { // first login: password only, then forced TOTP binding
		if pwOK {
			s.startSession(w, r, ip)
			http.Redirect(w, r, "/setup/totp", http.StatusSeeOther)
			return
		}

		step, codeOK := websec.VerifyTOTP(admin.TOTPSecret, r.PostFormValue("code"), s.now())

				s.notifyLogin(ip, r.UserAgent())
				http.Redirect(w, r, "/", http.StatusSeeOther)
				return
			}
